2855:     private fun issueAutoScrollStart(session: Int) {
2856:         // Pref is half-steps (speed x2); level is 1.0..10.0 in 0.5 increments.
2857:         val level = preferences.novelAutoScrollLevel()
2858: 
2859:         // Drive the scroll from a single in-page requestAnimationFrame loop instead of a Kotlin
2860:         // timer that fires window.scrollBy over the JS bridge every 50ms: those round-trips arrive
2861:         // with jittery timing and each moves a fixed integer step, which reads as stutter. The rAF
2862:         // loop advances by (px/sec * frame delta) with sub-pixel accumulation for smooth motion, and
2863:         // naturally pauses while the WebView is backgrounded (no frames), resuming without a jump via
2864:         // the dt clamp. speed level (1..10) maps to CSS px/sec.
2865:         val pxPerSec = level * 20
2866:         val attempt = ++autoScrollStartAttempt
2867:         evaluateJavascriptSafe(
2868:             """
2869:             (function() {
2870:                 var s = window.__tdAutoScroll || (window.__tdAutoScroll = {});
2871:                 s.pxPerSec = $pxPerSec;
2872:                 if (s.running) return;
2873:                 s.running = true;
2874:                 s.last = null;
2875:                 s.acc = 0;
2876:                 function step(ts) {
2877:                     if (!s.running) return;
2878:                     if (s.last === null) s.last = ts;
2879:                     var dt = (ts - s.last) / 1000;
2880:                     s.last = ts;
2881:                     // Clamp so a long background gap (or first frame) can't jump the page.
2882:                     if (dt > 0.05) dt = 0.05;
2883:                     s.acc += s.pxPerSec * dt;
2884:                     var whole = Math.floor(s.acc);
2885:                     if (whole > 0) { window.scrollBy(0, whole); s.acc -= whole; }
2886:                     s.raf = requestAnimationFrame(step);
2887:                 }
2888:                 s.raf = requestAnimationFrame(step);
2889:             })();
2890:             """.trimIndent(),
2891:             null,
2892:         )
